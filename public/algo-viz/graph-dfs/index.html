def dfs(graph, start):
  stack = [start]
  visited = set()
  while stack:
    node = stack.pop()
    if node in visited:
      continue
    visited.add(node)
    for nbr in graph[node]:
      if nbr not in visited:
        stack.append(nbr)
  return visited
