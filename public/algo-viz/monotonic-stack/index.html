def next_greater_element(nums):
    n = len(nums)
    result = [-1] * n
    stack = []  # stores indices
 
    for i in range(n):
        # pop while stack top < current (NGE)
        while stack and nums[stack[-1]] < nums[i]:
            idx = stack.pop()
            result[idx] = nums[i]
        stack.append(i)
 
    # remaining have no next greater element
    # result[idx] stays -1
    return result
