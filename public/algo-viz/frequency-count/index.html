def frequency_count(arr):


    freq = {}                       # init empty map


    for element in arr:


        if element in freq:


            freq[element] += 1       # increment


        else:


            freq[element] = 1       # first occurrence


    return freq


                                    


# find most frequent element


most_freq = max(freq, key=freq.get)
