quicksort(A, lo, hi):


  if lo < hi:


    p = partition(A, lo, hi)


    quicksort(A, lo, p-1)


    quicksort(A, p+1, hi)


 


partition(A, lo, hi):


  pivot = A[hi]


  i = lo - 1


  for j in [lo, hi):


    if A[j] <= pivot:


      i++


      swap(A[i], A[j])


  swap(A[i+1], A[hi])


  return i+1
