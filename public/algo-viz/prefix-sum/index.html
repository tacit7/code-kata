// Build prefix sum
prefix = [0] * (n+1)
for i in range(1, n+1):
  prefix[i] = prefix[i-1]
            + arr[i-1]
 
// Range Query  O(1)
def rangeSum(l, r):
  return prefix[r+1]
       - prefix[l]
