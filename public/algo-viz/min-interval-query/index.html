def minInterval(intervals, queries):
  # sort intervals by start
  ivs = sorted(intervals)
  qs = sorted(enumerate(queries),
              key=lambda x: x[1])
  heap = MinHeap()  # (size, end)
  ans = [-1] * len(queries)
  ptr = 0

  for origIdx, q in qs:
    # push intervals with start <= q
    while ptr < len(ivs) and ivs[ptr][0] <= q:
      s, e = ivs[ptr]
      heap.push((e-s+1, e))
      ptr += 1
    # evict expired (end < q)
    while heap and heap.peek()[1] < q:
      heap.pop()
    if heap:
      ans[origIdx] = heap.peek()[0]

  return ans
