def top_k_frequent(nums, k): count = {} for n in nums: count[n] = count.get(n, 0) + 1 freq = [[] for _ in range(len(nums) + 1)] for n, c in count.items(): freq[c].append(n) res = [] for i in range(len(freq) - 1, 0, -1): for n in freq[i]: res.append(n) if len(res) == k: return res
