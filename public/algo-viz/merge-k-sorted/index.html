def mergeKLists(lists):
  heap = MinHeap()
  for i, head in enumerate(lists):
    if head:
      heap.push((head.val,i,head))
  dummy = ListNode(0)
  cur = dummy
  while heap:
    val,i,node = heap.pop()
    cur.next = node
    cur = cur.next
    if node.next:
      heap.push((node.next.val,
                  i, node.next))
  return dummy.next
